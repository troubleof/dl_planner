cmake_minimum_required(VERSION 3.16)
project(pointcloud_nav)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters kdtree)
find_package(OpenCV REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(fmt REQUIRED)
foreach(dep rclcpp geometry_msgs nav_msgs sensor_msgs visualization_msgs message_filters cv_bridge pcl_conversions)
  find_package(${dep} REQUIRED)
endforeach()
file(GLOB_RECURSE ROG_SOURCES third_party/rog/src/*.cpp third_party/rog/include/*.cpp)
file(GLOB EGO_SOURCES third_party/ego/src/*.cpp)
add_executable(pointcloud_nav_worker src/worker.cpp src/maps.cpp ${EGO_SOURCES} ${ROG_SOURCES})
target_compile_definitions(pointcloud_nav_worker PRIVATE ORIGIN_AT_CORNER USE_ROS2 ROOT_DIR="./")
target_include_directories(pointcloud_nav_worker PRIVATE include third_party/ego/include third_party/rog/include ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
ament_target_dependencies(pointcloud_nav_worker rclcpp geometry_msgs nav_msgs sensor_msgs visualization_msgs message_filters cv_bridge pcl_conversions)
target_link_libraries(pointcloud_nav_worker ${PCL_LIBRARIES} ${OpenCV_LIBS} yaml-cpp fmt::fmt)
install(TARGETS pointcloud_nav_worker DESTINATION lib/${PROJECT_NAME})
add_executable(route_manager_check src/route_manager_check.cpp third_party/ego/src/dyn_a_star.cpp)
target_include_directories(route_manager_check PRIVATE include third_party/ego/include ${EIGEN3_INCLUDE_DIRS})
ament_target_dependencies(route_manager_check rclcpp)
target_link_libraries(route_manager_check yaml-cpp)
install(TARGETS route_manager_check DESTINATION lib/${PROJECT_NAME})
ament_package()
